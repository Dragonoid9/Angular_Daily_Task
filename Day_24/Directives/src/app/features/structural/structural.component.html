Using * for the Conditional

{{ a }} is greater than {{ b }}
{{ b }} is greater than {{ a }}

Using @ for the Conditional

@if(a>b){ {{ a }} is greater than {{ b }} }@else{ {{ b }} is greater than {{ a }} }

Using * for the ForLoop in the items

{{ item }}

Using @ for the ForLoop

@for(item of items; track items){
{{ item }}
}

This is second item.
This is the default item.

@switch(item){ @case("first"){

This item is first.

} @case("second") {

This item is second.

} @default{

This is the default item.
